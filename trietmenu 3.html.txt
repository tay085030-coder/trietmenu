// --- share helpers ---
const owner = 'tay085030-coder';
const repo = '-triet-shop-triet';
const filePath = 'trietmenu 3.html';
const branch = 'main'; // change if default branch differs

const blobUrl = `https://github.com/${owner}/${repo}/blob/${branch}/${encodeURIComponent(filePath)}`;
const rawUrl = `https://raw.githubusercontent.com/${owner}/${repo}/${branch}/${encodeURIComponent(filePath)}`;
const pagesUrl = `https://${owner}.github.io/${repo}/${encodeURIComponent(filePath)}`;

// Buttons
const btnNativeShare = document.getElementById('btnNativeShare');
const btnCopyBlob = document.getElementById('btnCopyBlob');
const btnCopyRaw = document.getElementById('btnCopyRaw');
const btnCopyPages = document.getElementById('btnCopyPages');
const btnTwitter = document.getElementById('btnTwitter');
const btnWhatsApp = document.getElementById('btnWhatsApp');

let currentTitle = document.title || 'TRIẾT GAMING';
let currentText = currentTitle + ' — Giao diện NEON V8';

async function tryClipboardWrite(text) {
 try {
   await navigator.clipboard.writeText(text);
   alert('Đã sao chép vào clipboard:\n' + text);
 } catch (e) {
   prompt('Sao chép thủ công, nhấn Ctrl/Cmd+C:', text);
 }
}

btnNativeShare.addEventListener('click', async ()=>{
 const shareData = { title: currentTitle, text: currentText, url: pagesUrl };
 if (navigator.share) {
   try {
     await navigator.share(shareData);
   } catch(e) {
     console.log('Share cancelled or failed', e);
   }
   return;
 }
 // fallback: copy blob link and open twitter/whatsapp
 await tryClipboardWrite(blobUrl);
 window.open(`https://twitter.com/intent/tweet?text=${encodeURIComponent(currentText + ' ' + blobUrl)}`,'_blank');
});

btnCopyBlob.addEventListener('click', ()=> tryClipboardWrite(blobUrl));
btnCopyRaw.addEventListener('click', ()=> tryClipboardWrite(rawUrl));
btnCopyPages.addEventListener('click', ()=> tryClipboardWrite(pagesUrl));

btnTwitter.addEventListener('click', ()=>{
 const url = `https://twitter.com/intent/tweet?text=${encodeURIComponent(currentText + ' ' + pagesUrl)}`;
 window.open(url,'_blank');
});

btnWhatsApp.addEventListener('click', ()=>{
 const url = `https://wa.me/?text=${encodeURIComponent(currentText + ' ' + pagesUrl)}`;
 window.open(url,'_blank');
});

// Make the page show usable link text somewhere if needed (console)
